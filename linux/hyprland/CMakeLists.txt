cmake_minimum_required(VERSION 3.30)
project(shiftdrop-hyprland VERSION 0.2.0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
if(NOT HYPRLAND_VERSION MATCHES "^0\\.56\\.2($|[+-])")
    message(FATAL_ERROR "ShiftDrop currently targets Hyprland 0.56.2; found ${HYPRLAND_VERSION}")
endif()
add_library(shiftdrop MODULE main.cpp)
set_target_properties(shiftdrop PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED YES)
target_link_libraries(shiftdrop PRIVATE PkgConfig::HYPRLAND)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(shiftdrop PRIVATE -fno-gnu-unique)
endif()
install(TARGETS shiftdrop LIBRARY DESTINATION lib)
